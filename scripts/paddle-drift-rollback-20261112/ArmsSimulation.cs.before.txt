using UnityEngine;

namespace Character.Control
{
    public class ArmsSimulation : MonoBehaviour
    {
        [SerializeField] private Rigidbody2D hammer;
        [SerializeField] private Transform hammerHead;
        [SerializeField] private Transform center;
        [SerializeField] private bool torqueEnabled = true;
        [field: SerializeField] public int MaxTorque { get; set; } = 1000;
        [SerializeField] private int torqueDampFactor = 1;
        [SerializeField] private ForceMode2D torqueMode = ForceMode2D.Force;
        [SerializeField] private bool forceEnabled = true;
        [SerializeField] private int maxHammerForce = 10;
        [SerializeField] private ForceMode2D hammerForceMode = ForceMode2D.Force;
        [field: SerializeField] public int MaxBodyForce { get; set; } = 50;
        [SerializeField] private ForceMode2D bodyForceMode = ForceMode2D.Force;

        // Airborne extension tracks radius independently of angular alignment.
        [SerializeField] private float extensionResponseTime = 0.12f;
        [SerializeField] private float maxExtensionSpeed = 3f;
        [SerializeField] private float extensionVelocityResponseTime = 0.08f;

        private float _supportedMass;
        private Rigidbody2D _rigidbody;
        private Vector2 _targetPosition;
        private bool _firstPositionChange;
        private bool _hammerHeadInContact;

        private void Awake()
        {
            _rigidbody = GetComponent<Rigidbody2D>();
            foreach (var body in transform.root.GetComponentsInChildren<Rigidbody2D>())
                if (body.bodyType == RigidbodyType2D.Dynamic) _supportedMass += body.mass;
            FindObjectOfType<TargetMovement>().OnPositionChanged += newPosition =>
            {
                _targetPosition = newPosition;
                _firstPositionChange = true;
            };
            var hammerHeadCollisionDetection = hammerHead.gameObject.GetComponentInChildren<CollisionDetection>();
            hammerHeadCollisionDetection.OnCollisionEnter += () => _hammerHeadInContact = true;
            hammerHeadCollisionDetection.OnCollisionExit += () => _hammerHeadInContact = false;
        }

        private void FixedUpdate()
        {
            if (!_firstPositionChange) return;

            var cauldronPosition = (Vector2) center.position;
            var hammerHeadPosition = (Vector2) hammerHead.position;
            var centerToHammerHead = cauldronPosition - hammerHeadPosition;
            var centerToTarget = cauldronPosition - _targetPosition;

            var angleDifference = Vector2.SignedAngle(centerToHammerHead, centerToTarget);
            if (Mathf.Abs(angleDifference) > 0.01 && torqueEnabled)
            {
                var torque = angleDifference * MaxTorque / 180f - _rigidbody.angularVelocity * torqueDampFactor;
                _rigidbody.AddTorque(torque, torqueMode);
            }

            var radialDifference = centerToHammerHead.magnitude - centerToTarget.magnitude;
            if (forceEnabled)
            {
                var direction = centerToHammerHead.normalized;
                if (_hammerHeadInContact)
                {
                    if (Mathf.Abs(radialDifference) <= 0.01f) return;
                    // The planted blade turns the same radius command into body reaction.
                    var desiredBodySpeed = Mathf.Clamp(
                        -radialDifference / Mathf.Max(0.01f, extensionResponseTime),
                        -maxExtensionSpeed, maxExtensionSpeed);
                    var relativeBodyVelocity = _rigidbody.GetPointVelocity(cauldronPosition)
                        - hammer.GetPointVelocity(hammerHeadPosition);
                    var bodySpeed = Vector2.Dot(relativeBodyVelocity, direction);
                    var speedError = desiredBodySpeed - bodySpeed;
                    // A completed outward stroke must not brake an already rising body.
                    if (radialDifference < 0f) speedError = Mathf.Max(0f, speedError);
                    var bodyForce = Mathf.Clamp(
                        _supportedMass * speedError / Mathf.Max(0.01f, extensionVelocityResponseTime),
                        -MaxBodyForce, MaxBodyForce);
                    _rigidbody.AddForce(direction * bodyForce, ForceMode2D.Force);
                }
                else
                {
                    // Use shaft direction even when the tip is retracted to the pivot.
                    var inward = -(Vector2)hammer.transform.right;
                    var desiredSpeed = Mathf.Clamp(
                        radialDifference / Mathf.Max(0.01f, extensionResponseTime),
                        -maxExtensionSpeed, maxExtensionSpeed);
                    var relativeVelocity = hammer.GetPointVelocity(hammerHeadPosition)
                        - _rigidbody.GetPointVelocity(cauldronPosition);
                    var speed = Vector2.Dot(relativeVelocity, inward);
                    var force = hammer.mass * (desiredSpeed - speed)
                        / Mathf.Max(0.01f, extensionVelocityResponseTime);
                    force = Mathf.Clamp(force, -maxHammerForce, maxHammerForce);
                    hammer.AddForce(inward * force, ForceMode2D.Force);
                }
            }
        }
    }
}
